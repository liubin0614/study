
```js
// 定义一个对象并为其实现同步迭代器
const myIterable = {
  items: [1, 2, 3, 4, 5],

  // 使用 Symbol.syncIterator 来实现同步迭代器
  [Symbol.syncIterator]() {
    let index = 0;
    const items = this.items;

    // 返回一个迭代器对象
    return {
      next() {
        if (index < items.length) {
          return { value: items[index++], done: false };
        } else {
          return { value: undefined, done: true };
        }
      }
    };
  }
}; 

// 使用 for...of 循环来迭代对象
for (const item of myIterable) {
  console.log(item); // 输出：1, 2, 3, 4, 5
}

```

```js
const asyncIterable = {
  // 异步数据源
  async *[Symbol.asyncIterator]() {
    const data = [1, 2, 3, 4, 5];
    
    for (const num of data) {
      // 模拟异步操作，例如网络请求
      await new Promise(resolve => setTimeout(resolve, 1000)); // 每次等待1秒
      yield num; // 返回异步迭代的下一个值
    }
  }
};

// 使用 for await...of 来异步迭代
(async () => {
  for await (const item of asyncIterable) {
    console.log(item); // 依次输出：1, 2, 3, 4, 5，每秒输出一个值
  }
})();
```
